1/**
2 * Definition for a binary tree node.
3 * struct TreeNode {
4 *     int val;
5 *     TreeNode *left;
6 *     TreeNode *right;
7 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
8 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
9 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left),
10 * right(right) {}
11 * };
12 */
13class Solution {
14public:
15    int cnt = 0;
16    int findDominantNodes(TreeNode* root) {
17        if (root == NULL)
18            return INT_MIN;
19
20        int l = findDominantNodes(root->left);
21        int r = findDominantNodes(root->right);
22
23        int mx=max(root->val,max(l,r));
24
25        if(mx==root->val)
26        {
27            cnt++;
28        }
29        return mx;
30    }
31    int countDominantNodes(TreeNode* root) {
32
33        if (root == NULL)
34            return 0;
35        findDominantNodes(root);
36
37        return cnt;
38    }
39};
